# Python Worker：使用 Upstash 官方 REST API（增强日志版）
# ---------------------------------------------------------

import json
import time
import os
from dotenv import load_dotenv
from upstash_redis import Redis

# 加载 .env 环境变量
load_dotenv()

# 初始化 Upstash Redis（REST 客户端）
redis = Redis(
    url=os.getenv("UPSTASH_REDIS_REST_URL"),
    token=os.getenv("UPSTASH_REDIS_REST_TOKEN")
)

print("Python Worker 已启动，等待任务...\n")

def pretty(obj):
    """结构化打印 JSON"""
    return json.dumps(obj, indent=2, ensure_ascii=False)

while True:
    try:
        # 从队列右侧取任务
        task_json = redis.rpop("task_queue")

        if task_json:
            task = json.loads(task_json)

            print("📥 收到任务：")
            print(pretty(task))

            task_id = task["task_id"]
            a = task["a"]
            b = task["b"]

            # 执行任务（示例：减法）
            result_value = a - b

            result_obj = {
                "status": "done",
                "result": {
                    "task_id": task_id,
                    "result": result_value
                }
            }

            print("\n🧮 任务执行结果：")
            print(pretty(result_obj))

           # 写回 Redis（一次性原子写入）
            final_result = {
                "status": "done",
                "result": result_value
            }

            redis.set(f"task_result:{task_id}", json.dumps(final_result))

            print("\n📤 已写回 Redis（原子写入）：")
            print(f"task_result:{task_id}")
            print(pretty(final_result))
            print("\n----------------------------------------\n")


        else:
            time.sleep(1)

    except Exception as e:
        print("❌ Worker 发生错误：", str(e))
        time.sleep(1)

// index.js
require("dotenv").config();
const express = require("express");
const { v4: uuidv4 } = require("uuid");
const { Redis } = require("@upstash/redis");

const app = express();
app.use(express.json());

const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL,
  token: process.env.UPSTASH_REDIS_REST_TOKEN,
});

function pretty(obj) {
  return JSON.stringify(obj, null, 2);
}

// -------------------------------
// 1）提交任务
// -------------------------------
app.post("/task/submit", async (req, res) => {
  try {
    console.log("\n📥 收到前端提交任务：");
    console.log(pretty(req.body));

    const { a, b } = req.body;
    const task_id = uuidv4();

    const task = { task_id, a, b };

    console.log("\n📤 推入 Redis 队列：task_queue");
    console.log(pretty(task));

    await redis.lpush("task_queue", JSON.stringify(task));

    const response = { status: "submitted", task_id };

    console.log("\n📨 返回给插件：");
    console.log(pretty(response));
    console.log("\n----------------------------------------\n");

    res.json(response);

  } catch (err) {
    console.error("❌ 提交任务接口发生错误：", err);
    res.status(500).json({ error: "Internal Server Error" });
  }
});

// -------------------------------
// 2）查询任务结果（最终正确版）
// -------------------------------
app.get("/task/result/:task_id", async (req, res) => {
  try {
    const { task_id } = req.params;

    console.log(`\n🔍 查询任务结果：task_result:${task_id}`);

    // ⭐ Upstash SDK 已自动 JSON.parse，不要再 JSON.parse
    const data = await redis.get(`task_result:${task_id}`);

    console.log("📦 Redis 返回：");
    console.log(pretty(data));

    // 任务未完成
    if (!data) {
      const pending = { status: "pending", result: null };
      console.log("⏳ 任务未完成：");
      console.log(pretty(pending));
      console.log("\n----------------------------------------\n");
      return res.json(pending);
    }

    // 任务已完成
    const response = {
      status: data.status,
      result: data.result,
    };

    console.log("✅ 任务已完成：");
    console.log(pretty(response));
    console.log("\n----------------------------------------\n");

    res.json(response);

  } catch (err) {
    console.error("❌ 查询任务接口发生错误：", err);
    res.status(500).json({ error: "Internal Server Error" });
  }
});

// -------------------------------
// 启动服务
// -------------------------------
app.listen(3000, () => {
  console.log("Node API 已启动：http://localhost:3000\n");
});

const output = vscode.window.createOutputChannel("AlphaPilot Logs");

export function activate(context: vscode.ExtensionContext) {
    output.appendLine("🚀 AlphaPilot 插件已启动");

    // 注册命令
    let disposable = vscode.commands.registerCommand(
        "alphaMinimalExtension.sendTask",
        async () => {
            try {
                const taskData = { a: 8, b: 2 };
                output.appendLine("📤 提交任务：" + JSON.stringify(taskData));

                // 调用 Node API 提交任务
                const submitResult = await fetch("http://localhost:3000/task/submit", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(taskData),
                });

                const submitJson = await submitResult.json();
                output.appendLine("📨 后端返回：" + JSON.stringify(submitJson));

                // 开始轮询
                pollTaskResult(submitJson.task_id);

            } catch (err: any) {
                output.appendLine("❌ 提交任务失败：" + err.message);
                vscode.window.showErrorMessage("提交任务失败：" + err.message);
            }
        }
    );

    context.subscriptions.push(disposable);
}

// ---------------------------------------------------------
// 轮询任务结果
// ---------------------------------------------------------
async function pollTaskResult(taskId: string) {
    const interval = setInterval(async () => {
        try {
            const res = await fetch(`http://localhost:3000/task/result/${taskId}`);
            const result: any = await res.json();

            output.appendLine("🔄 轮询结果：" + JSON.stringify(result));

            if (result.status === "done") {
                clearInterval(interval);

                output.appendLine("✅ 任务完成：" + JSON.stringify(result));
                vscode.window.showInformationMessage(
                    `任务完成，结果为：${result.result}`
                );
            }

        } catch (err: any) {
            clearInterval(interval);
            output.appendLine("❌ 轮询失败：" + err.message);
            vscode.window.showErrorMessage("轮询失败：" + err.message);
        }
    }, 1000);
}

export function deactivate() {
    output.appendLine("🛑 AlphaPilot 插件已停用");
}
